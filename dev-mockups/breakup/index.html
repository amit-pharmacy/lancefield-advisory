Breaking up the long pages

Three changes from one CSS + JS snippet. They work on every category (hub) and subcategory page, so you set them up once, then add one class to the section headings on each page. Below are exact copies of two live pages with the changes applied and tagged in pink.

The three changes

All three come from the same paste. If you only do one, do number 2.

1

Hide the decorative panels on phones

On a phone the panel images stack into the middle of the text, about 350px each, and carry no information. Tablet and desktop keep them, where they sit beside the text.

Effort: none beyond pasting the CSS. It finds any image whose file name starts with panel- and hides it, plus the empty column it sat in.

2

Number the sections and add an "On this page" box

Each section heading gets a small number (01, 02...) and a "Contents ↑" link, and a contents box after the intro lists every section. A 20-screen page becomes a menu: people jump to the part they came for, and every section has a clear start.

Effort: paste the JS, then add the CSS class la-chapter to each section heading (step 3 below). The script builds the numbers and the box from those headings, so new pages just work.

3

One text style on phones

Right now body text on these pages comes in 10 size and colour combinations and starts anywhere from 20px to 50px from the screen edge. Proposed: 16px dark body text, the first paragraph under each heading as an 18px navy lead, and one 24px edge. Cards keep their inner padding.

Effort: none beyond pasting the CSS. It overrides the per-section padding and text settings on phones only.

How to add it

  1. Paste la-breakup.css into Elementor > Site Settings > Custom CSS (or Appearance > Customise > Additional CSS).
  2. Paste la-breakup.js into Elementor > Custom Code, location </body> - End, entire site. A footer snippet in WPCode works too. Elementor Custom Code needs it wrapped in <script> ... </script>.
  3. On each hub and subpage, click each section heading > Advanced > CSS Classes > add la-chapter. Do it while you change those headings from H1 to H2. Skip the hero title, the dark CTA bands, the pull quote and "Related pages".
  4. Check on a phone: numbers above each section, the contents box after the intro, no panel images, text lined up on one edge.

la-breakup.jsDownload
/* Lancefield: break up long hub and subpages. Paste once, site-wide, before </body>. */
(function () {
  function lum(c) { var m = c.match(/[\d.]+/g); return m ? 0.299 * m[0] + 0.587 * m[1] + 0.114 * m[2] : 255; }
  function clear(c) { var m = c.match(/[\d.]+/g); return !m || (m.length > 3 && +m[3] === 0); }
  function slug(t) { return t.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '').slice(0, 60); }
  function esc(t) { return t.replace(/&/g, '&amp;').replace(/</g, '&lt;'); }
  function bgOf(el) { while (el) { var c = getComputedStyle(el).backgroundColor; if (!clear(c)) return c; el = el.parentElement; } return 'rgb(255, 255, 255)'; }

  function run() {
    var page = document.querySelector('[data-elementor-type="wp-page"]');
    if (!page) return;

    // MOVE 3: tag dark body text on light backgrounds, and boxes that are real cards
    // (own background colour, a shadow or a left bar) so they keep their inner padding
    page.querySelectorAll('.elementor-widget-text-editor, .elementor-icon-box-description').forEach(function (el) {
      if (lum(getComputedStyle(el).color) < 140) el.classList.add('la-body');
    });
    page.querySelectorAll('.e-con .e-con').forEach(function (c) {
      var s = getComputedStyle(c);
      var ownBg = !clear(s.backgroundColor) && s.backgroundColor !== bgOf(c.parentElement);
      if (ownBg || s.boxShadow !== 'none' || parseFloat(s.borderLeftWidth)) c.classList.add('la-card');
    });

    // MOVE 2: number every heading tagged la-chapter and build the contents box
    var heads = page.querySelectorAll('.la-chapter');
    if (heads.length < 3) return;
    var items = '';
    heads.forEach(function (w, i) {
      var n = (i < 9 ? '0' : '') + (i + 1);
      var title = w.textContent.trim();
      if (!w.id) w.id = slug(title);
      items += '<li><a href="#' + w.id + '"><span>' + n + '</span>' + esc(title) + '</a></li>';
      var mark = document.createElement('div');
      mark.className = 'la-mark';
      mark.innerHTML = '<span>' + n + '</span><i></i><a href="#on-this-page">Contents &uarr;</a>';
      w.insertBefore(mark, w.firstChild);
      // MOVE 3: first paragraph straight after the heading becomes the lead
      var next = w.nextElementSibling;
      if (next && next.classList.contains('elementor-widget-text-editor')) (next.querySelector('p') || next).classList.add('la-lead');
    });
    var band = heads[0];
    while (band.parentElement !== page) band = band.parentElement;
    var toc = document.createElement('div');
    toc.className = 'la-toc-band';
    toc.innerHTML = '<nav class="la-toc" id="on-this-page" aria-label="On this page"><p>On this page</p><ol>' + items + '</ol></nav>';
    page.insertBefore(toc, band);
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', run); else run();
})();
